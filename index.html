<input type="text" id="answer" placeholder="輸入答案">

<br>

<button onclick="checkAnswer()">提交</button>

<div id="result"></div>

<script>

let vocab = [];

let queue = [];

let current = null;

let direction = "";

document.getElementById("fileInput").addEventListener("change", function(e) {

  const reader = new FileReader();

  reader.onload = function() {

    vocab = reader.result

      .trim()

      .split("\n")

      .map(row => row.split(","));

    queue = [...vocab];

    nextQuestion();

  };

  reader.readAsText(e.target.files[0], "UTF-8");

});

function nextQuestion() {

  if (queue.length === 0) {

    document.getElementById("card").innerText = "🎉 全部答啱晒！";

    return;

  }

  const index = Math.floor(Math.random() * queue.length);

  current = queue[index];

  direction = Math.random() > 0.5 ? "CN" : "JP";

  document.getElementById("card").innerText =

    direction === "CN" ? current[0] : current[1];

  document.getElementById("answer").value = "";

  document.getElementById("result").innerText = "";

}

function checkAnswer() {

  const input = document.getElementById("answer").value.trim();

  const correct = direction === "CN" ? current[1] : current[0];

  if (input === correct) {

    document.getElementById("result").innerText = "✅ 啱咗！";

    queue = queue.filter(item => item !== current);

  } else {

    document.getElementById("result").innerText =

      "❌ 錯咗！正確答案係：" + correct;

  }

  setTimeout(nextQuestion, 1000);

}

</script>
